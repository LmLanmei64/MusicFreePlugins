async function getMusicComments(musicItem, page = 1) {
    const pageSize = 20; // 每页评论数量
    const url = "https://163api.qijieya.cn/comment/music";
    try {
        const response = await axios.get(url, {
            params: {
                id: musicItem.id,
                page: page,
                pageSize: pageSize
            }
        });
        const result = response.data;
        // 假设返回结构为 { comments: [], hasMore: boolean }
        const comments = (result.comments || []).map(formatComment);
        return {
            isEnd: !result.hasMore,
            data: comments
        };
    } catch (error) {
        console.error("获取评论时出错:", error.message);
        return {
            isEnd: true,
            data: []
        };
    }
}
